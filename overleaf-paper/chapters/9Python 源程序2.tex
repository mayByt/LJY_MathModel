class BearingDataset(Dataset):
    def __init__(self, csv_files, selected_features, label_encoder=None, frac=1.0, scaler=None):
        dfs = []
        for f in csv_files:
            df = pd.read_csv(f)
            if frac < 1.0:
                df = df.sample(frac=frac, random_state=42)
            dfs.append(df)
        df = pd.concat(dfs, ignore_index=True)

        # # 统计并过滤 BA 样本
        # total_samples = len(df)
        # ba_samples = df["Label"].str.contains("BA").sum()
        # df = df[~df["Label"].str.contains("BA")]
        # print(f"[BearingDataset] Total samples: {total_samples}, "
        #       f"Filtered BA samples: {ba_samples}, "
        #       f"Remaining: {len(df)}")

        total_samples = len(df)
        ba_samples = df["Label"].str.contains("BA").sum()

        # 构建筛选条件
        mask_ba = df["Label"].str.contains("BA")
        mask_de_fe = df["Label"].str.startswith("DE") & df["Label"].str.endswith("FE")
        mask_fe_de = df["Label"].str.startswith("FE") & df["Label"].str.endswith("DE")

        # 统计数量
        de_fe_samples = mask_de_fe.sum()
        fe_de_samples = mask_fe_de.sum()

        # 过滤掉这三类
        df = df[~(mask_ba | mask_de_fe | mask_fe_de)]

        print(f"[BearingDataset] Total samples: {total_samples}, "
              f"Filtered BA: {ba_samples}, "
              f"Filtered DE..FE: {de_fe_samples}, "
              f"Filtered FE..DE: {fe_de_samples}, "
              f"Remaining: {len(df)}")


        df["FaultType"] = df["Label"].apply(lambda x: x.split("_")[1])
        df["Channel"] = df["Channel"].astype(str)
        feature_cols = selected_features

        if scaler is None:
            self.scaler = StandardScaler()
            df[feature_cols] = self.scaler.fit_transform(df[feature_cols])
        else:
            self.scaler = scaler
            df[feature_cols] = self.scaler.transform(df[feature_cols])

        self.X, self.y = [], []
        for _, row in df.iterrows():
            feats = row[feature_cols].values.astype(np.float32)
            self.X.append(feats)
            self.y.append(row["FaultType"])

        self.X = np.array(self.X, dtype=np.float32)
        self.y = np.array(self.y)

        if label_encoder is None:
            self.label_encoder = LabelEncoder()
            self.y = self.label_encoder.fit_transform(self.y)
        else:
            self.label_encoder = label_encoder
            self.y = self.label_encoder.transform(self.y)

    def __len__(self):
        return len(self.X)

    def __getitem__(self, idx):
        return torch.tensor(self.X[idx], dtype=torch.float32), torch.tensor(self.y[idx], dtype=torch.long)

class BearingNet(nn.Module):
    def __init__(self, num_features, num_classes):
        super(BearingNet, self).__init__()
        self.net = nn.Sequential(
            nn.Linear(num_features, 128),
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(128, 64),
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(64, num_classes)
        )

    def forward(self, x):
        return self.net(x)

def train_model(train_loader, test_loader, model, device, epochs=200, lr=5e-4, patience=20,
                log_file=None, save_path="/home/xxxten/PycharmProjects/pythonProject/model/MLP/log_no_mean/best_model.pth"):
    criterion = nn.CrossEntropyLoss()
    optimizer = optim.Adam(model.parameters(), lr=lr)
    scheduler = optim.lr_scheduler.StepLR(optimizer, step_size=50, gamma=0.5)

    best_test_acc = 0.0
    patience_counter = 0

    for epoch in range(epochs):
        model.train()
        total_loss, correct, total = 0.0, 0, 0
        for X, y in train_loader:
            X, y = X.to(device), y.to(device)
            optimizer.zero_grad()
            outputs = model(X)
            loss = criterion(outputs, y)
            loss.backward()
            optimizer.step()

            total_loss += loss.item()
            _, pred = outputs.max(1)
            correct += pred.eq(y).sum().item()
            total += y.size(0)

        train_acc = correct / total

        # Test
        model.eval()
        correct, total = 0, 0
        test_loss = 0.0
        with torch.no_grad():
            for X, y in test_loader:
                X, y = X.to(device), y.to(device)
                outputs = model(X)
                loss = criterion(outputs, y)
                test_loss += loss.item()
                _, pred = outputs.max(1)
                correct += pred.eq(y).sum().item()
                total += y.size(0)

        test_acc = correct / total

        log_str = (f"Epoch [{epoch+1}/{epochs}] "
                   f"Train Loss: {total_loss/len(train_loader):.4f} "
                   f"Train Acc: {train_acc:.4f} "
                   f"Test Loss: {test_loss/len(test_loader):.4f} "
                   f"Test Acc: {test_acc:.4f} "
                   f"LR: {scheduler.get_last_lr()[0]:.6f}")

        print(log_str)
        if log_file:
            with open(log_file, "a") as f:
                f.write(log_str + "\n")

        scheduler.step()

        # Early stopping
        if test_acc > best_test_acc:
            best_test_acc = test_acc
            patience_counter = 0
            torch.save(model.state_dict(), save_path)  # 保存最优模型
            if log_file:
                with open(log_file, "a") as f:
                    f.write(f"Saved best model at epoch {epoch+1} with Test Acc: {best_test_acc:.4f}\n")
        else:
            patience_counter += 1
            if patience_counter >= patience:
                stop_str = f"Early stopping at epoch {epoch+1}"
                print(stop_str)
                if log_file:
                    with open(log_file, "a") as f:
                        f.write(stop_str + "\n")
                break

    print(f"Best Test Accuracy: {best_test_acc:.4f}")
    if log_file:
        with open(log_file, "a") as f:
            f.write(f"Best Test Accuracy: {best_test_acc:.4f}\n")

def test_model(test_loader, model, device, criterion=None):
    model.eval()
    correct, total = 0, 0
    total_loss = 0.0
    with torch.no_grad():
        for X, y in test_loader:
            X, y = X.to(device), y.to(device)
            outputs = model(X)
            if criterion:
                total_loss += criterion(outputs, y).item()
            _, pred = outputs.max(1)
            correct += pred.eq(y).sum().item()
            total += y.size(0)
    acc = correct / total
    avg_loss = total_loss / len(test_loader) if criterion else None
    return acc, avg_loss
